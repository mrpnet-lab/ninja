```csharp
private double CalcPreEntryTrapScore(int direction)
{
    if (CurrentBar < 5) return 0;
    double score = 0;
    double atr = indAtr[0];
    if (atr <= 0) return 0;

    // Recent adverse move against intended direction
    double recentMove = direction == 1
        ? Close[0] - Math.Min(Low[0], Low[1])   // longs: recent drop
        : Math.Max(High[0], High[1]) - Close[0]; // shorts: recent rise
    double atrPts = atr / (TickSize * NQ_TICKS_PER_POINT);
    if (atrPts > 0) score += Math.Min(40, (recentMove / (TickSize * NQ_TICKS_PER_POINT)) / atrPts * 30);

    // EMA opposing direction
    if (direction == 1  && indEmaFast[0] < indEmaSlow[0]) score += 25;
    if (direction == -1 && indEmaFast[0] > indEmaSlow[0]) score += 25;

    // Wick rejection against direction on current bar
    double barRange = High[0] - Low[0];
    if (barRange > 0)
    {
        if (direction == 1)  { double uw = (High[0] - Math.Max(Close[0], Open[0])) / barRange; if (uw > 0.45) score += 20; }
        if (direction == -1) { double lw = (Math.Min(Close[0], Open[0]) - Low[0])  / barRange; if (lw > 0.45) score += 20; }
    }
    return Math.Min(100, score);
}
```

### UpdateVolumeProfile() — call from OnBarUpdate() every bar:

```csharp
private void UpdateVolumeProfile()
{
    double roundedPrice = Math.Round(Close[0] / TickSize) * TickSize;
    double vol = Volume[0];
    if (vol <= 0) return;

    if (volumeAtPrice.ContainsKey(roundedPrice))
        volumeAtPrice[roundedPrice] += vol;
    else
        volumeAtPrice[roundedPrice] = vol;

    totalSessionVolume += vol;

    if (volumeAtPrice.Count < 5 || totalSessionVolume <= 0) return;

    // Find POC
    pocLevel = 0;
    double maxVol = 0;
    foreach (var kvp in volumeAtPrice)
        if (kvp.Value > maxVol) { maxVol = kvp.Value; pocLevel = kvp.Key; }

    // Calculate VAH/VAL (70% of volume)
    var sorted = volumeAtPrice.OrderByDescending(k => k.Value).ToList();
    double cumVol = 0;
    double target = totalSessionVolume * 0.70;
    var valueAreaPrices = new List<double>();
    foreach (var kvp in sorted)
    {
        cumVol += kvp.Value;
        valueAreaPrices.Add(kvp.Key);
        if (cumVol >= target) break;
    }
    if (valueAreaPrices.Count > 0)
    {
        vahLevel = valueAreaPrices.Max();
        valLevel = valueAreaPrices.Min();
        volumeProfileReady = true;
    }
}
```
