    int barDir = Close[0] >= Open[0] ? 1 : -1;
    trapDirectionHistory.Enqueue(barDir);
    while (trapDirectionHistory.Count > trapDetectWindowBars)

    if (trapDirectionHistory.Count < trapDetectWindowBars / 2) return; // not enough data

Step 2 — COUNT REVERSALS:
    var dirs = trapDirectionHistory.ToArray();
    int reversals = 0;
    for (int i = 1; i < dirs.Length; i++)
        if (dirs[i] != dirs[i-1]) reversals++;
    double priceRange = MAX(High, trapDetectWindowBars)[0] - MIN(Low, trapDetectWindowBars)[0];
    double rangeInPoints = priceRange / (TickSize * NQ_TICKS_PER_POINT);

Step 3 — COMPUTE VOLUME AVERAGE:
    double volSum = 0;
    int vlb = Math.Min(20, CurrentBar - 1);
    for (int i = 1; i <= vlb; i++) volSum += Volume[i];
    double avgVol20 = vlb > 0 ? volSum / vlb : Volume[0];

Step 4 — TRAP PATTERN RECOGNITION:
    bool highReversals  = reversals >= (int)(trapDetectWindowBars * 0.6);
    bool tightRange     = rangeInPoints < indAtr[0] / (TickSize * NQ_TICKS_PER_POINT) * 4.0;
    bool lowVolume      = avgVol20 > 0 && Volume[0] < avgVol20 * 0.70;
    bool inTradeEnough  = openDcaCount > 0 && (DateTime.Now - lastEntryWallTime).TotalSeconds > 30;

    if (highReversals && tightRange && lowVolume && inTradeEnough && !trapPatternActive)
    {
        trapPatternActive      = true;
        trapPatternBarsActive  = 0;
        trapAdaptiveMode       = 1;   // Mode 1: protect
        trapRangeHigh          = MAX(High, trapDetectWindowBars)[0];
        trapRangeLow           = MIN(Low,  trapDetectWindowBars)[0];
        trapOriginalTP         = hiddenTargetPrice;
        // MODE 1: Move SL to breakeven
        double beTick          = TickSize;
        if (openTradeDirection == 1)  hiddenStopPrice = averageEntryPrice + beTick;
        else                          hiddenStopPrice = averageEntryPrice - beTick;
        Print("POST-ENTRY TRAP MODE 1 ACTIVATED: " + reversals + " reversals, range=" 
            + rangeInPoints.ToString("F1") + "pts, vol=" 
            + (Volume[0]/avgVol20*100).ToString("F0") + "% of avg. SL→Breakeven.");
    }
    else if (trapPatternActive)
    {
        trapPatternBarsActive++;
        
        // MODE 2: Range trade — activate after 5 confirmed bars in trap
        if (trapAdaptiveMode == 1 && trapPatternBarsActive >= 5)
        {
            trapAdaptiveMode = 2;
            double rangeTop    = trapRangeHigh;
            double rangeBottom = trapRangeLow;
            double rangePct    = trapRangeHigh - trapRangeLow > 0
                ? (Close[0] - trapRangeLow) / (trapRangeHigh - trapRangeLow) : 0.5;
            
            if (openTradeDirection == 1 && rangePct > 0.75)
                hiddenTargetPrice = trapRangeHigh - 2 * TickSize;   // target range top
            else if (openTradeDirection == -1 && rangePct < 0.25)
                hiddenTargetPrice = trapRangeLow  + 2 * TickSize;   // target range bottom
            Print("POST-ENTRY TRAP MODE 2: targeting range edge @ " 
                + hiddenTargetPrice.ToString("F2"));
        }
        
        // EXPIRY: if pattern weakens for 3 consecutive bars, cancel trap mode
        if (reversals < (int)(trapDetectWindowBars * 0.4))
        {
            trapExpiryCandidateBars++;
            if (trapExpiryCandidateBars >= 3)
            {
                trapPatternActive      = false;
                trapAdaptiveMode       = 0;
                trapExpiryCandidateBars = 0;
                if (trapOriginalTP > 0) hiddenTargetPrice = trapOriginalTP;
                Print("Post-entry trap pattern expired — TP restored to " 
                    + hiddenTargetPrice.ToString("F2"));
            }
        }
        else { trapExpiryCandidateBars = 0; }
    }
